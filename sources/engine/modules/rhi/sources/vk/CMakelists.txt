
find_package(Vulkan REQUIRED)

if(Vulkan_FOUND)
	message(STATUS "Vulkan found: ${Vulkan_INCLUDE_DIRS}")
	target_include_directories(${Target_Name} PRIVATE ${Vulkan_INCLUDE_DIRS})
	target_link_libraries(${Target_Name} PRIVATE ${Vulkan_LIBRARIES})
else()
	message(FATAL_ERROR "Vulkan not found")
endif()


# 专门伺候 <windows.h> 这个头文件:
# VK_USE_PLATFORM_WIN32_KHR: 用于启动 win32 surface
# NOMINMAX: 避免 windows.h 定义 min/max 宏
# WIN32_LEAN_AND_MEAN: 避免 windows.h 引入过多不必要的头文件
if (WIN32)
	target_compile_definitions(${Target_Name} PRIVATE VK_USE_PLATFORM_WIN32_KHR NOMINMAX WIN32_LEAN_AND_MEAN)
endif()



target_link_libraries(${Target_Name} 
	PUBLIC 
		Vulkan::Vulkan
)
